cmake_minimum_required(VERSION 3.16)
project(lr_cpu LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
include_directories(${ROOT}/gotools-core/include ${ROOT}/lrsplines2D/include ${ROOT}/viewlib/include ${ROOT}/sisl/include ${ROOT})
file(GLOB SISL CONFIGURE_DEPENDS ${ROOT}/sisl/src/*.c)
file(GLOB_RECURSE CORE CONFIGURE_DEPENDS ${ROOT}/gotools-core/src/*.C ${ROOT}/gotools-core/src/*.cpp)
file(GLOB LR CONFIGURE_DEPENDS ${ROOT}/lrsplines2D/src/*.C ${ROOT}/lrsplines2D/src/*.cpp)
list(FILTER LR EXCLUDE REGEX "/(LogLikelyhood|LRBenchmarkUtils)\\.C$")
add_library(sisl STATIC ${SISL})
add_library(core STATIC ${CORE})
add_library(lr STATIC ${LR})
add_library(lrmesh STATIC ${ROOT}/viewlib/src/vol_and_lr/LRSurfaceTesselator.C)
target_link_libraries(core PUBLIC sisl)
target_link_libraries(lr PUBLIC core)
target_link_libraries(lrmesh PUBLIC lr)
add_executable(lr_baseline ${ROOT}/task_support/baseline.cpp)
target_link_libraries(lr_baseline PRIVATE lrmesh)
if(TASK_TARGET)
 if(NOT EXISTS "${ROOT}/tests/lr_coarsen.cpp")
  message(FATAL_ERROR "tests/lr_coarsen.cpp missing")
 endif()
 set_source_files_properties(${ROOT}/task_support/driver.c PROPERTIES LANGUAGE CXX)
 add_executable(target ${ROOT}/tests/lr_coarsen.cpp ${ROOT}/task_support/driver.c)
 target_sources(target PRIVATE ${ROOT}/lrsplines2D/src/LRCoarsen.C)
 target_link_libraries(target PRIVATE lrmesh)
 add_executable(lr_coarsen_obj ${ROOT}/examples/lr_coarsen_obj.cpp)
 target_link_libraries(lr_coarsen_obj PRIVATE lrmesh)
endif()
