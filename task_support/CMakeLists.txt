cmake_minimum_required(VERSION 3.16)
project(session_support LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB core CONFIGURE_DEPENDS "../opensubdiv/sdc/*.cpp" "../opensubdiv/vtr/*.cpp" "../opensubdiv/far/*.cpp" "../opensubdiv/bfr/*.cpp")
add_library(kernel STATIC ${core} ../opensubdiv/version.cpp)
target_compile_definitions(kernel PUBLIC OPENSUBDIV_GREGORY_EVAL_TRUE_DERIVATIVES)
target_include_directories(kernel PUBLIC ..)
add_executable(baseline baseline.cpp)
target_link_libraries(baseline PRIVATE kernel)
