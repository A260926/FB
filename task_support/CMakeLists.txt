cmake_minimum_required(VERSION 3.15)
project(PhotonBeamTaskSupport LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(EMBREE_TUTORIALS OFF CACHE BOOL "" FORCE)
set(EMBREE_ISPC_SUPPORT OFF CACHE BOOL "" FORCE)
set(EMBREE_SYCL_SUPPORT OFF CACHE BOOL "" FORCE)
set(EMBREE_TASKING_SYSTEM INTERNAL CACHE STRING "" FORCE)
set(EMBREE_RAY_PACKETS OFF CACHE BOOL "" FORCE)
set(EMBREE_STATIC_LIB ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
foreach(geometry QUAD CURVE SUBDIVISION USER INSTANCE INSTANCE_ARRAY GRID POINT)
  set(EMBREE_GEOMETRY_${geometry} OFF CACHE BOOL "" FORCE)
endforeach()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  set(EMBREE_MAX_ISA NEON CACHE STRING "" FORCE)
else()
  set(EMBREE_MAX_ISA SSE2 CACHE STRING "" FORCE)
endif()
get_filename_component(TASK_REPO "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
add_subdirectory("${TASK_REPO}" core)
add_executable(taskbaseline baseline.cpp)
target_link_libraries(taskbaseline PRIVATE embree)
if(TASK_TARGET)
  set_source_files_properties(driver.c PROPERTIES LANGUAGE CXX)
  add_executable(tasktarget "${TASK_REPO}/task_tests/beams.cpp" driver.c api.c "${TASK_REPO}/tutorials/photon_beams/renderer.cpp")
  target_include_directories(tasktarget PRIVATE "${TASK_REPO}")
  target_compile_options(tasktarget PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-include${TASK_REPO}/tutorials/photon_beams/renderer.h>")
  target_link_libraries(tasktarget PRIVATE embree)
  set_target_properties(tasktarget PROPERTIES OUTPUT_NAME target)
  add_executable(taskexample "${TASK_REPO}/examples/photon_beams.cpp" "${TASK_REPO}/tutorials/photon_beams/renderer.cpp")
  target_include_directories(taskexample PRIVATE "${TASK_REPO}")
  target_compile_options(taskexample PRIVATE "-include${TASK_REPO}/tutorials/photon_beams/renderer.h")
  target_link_libraries(taskexample PRIVATE embree)
endif()
