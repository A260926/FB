cmake_minimum_required(VERSION 3.16)
project(trim_mesh_support LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB_RECURSE kernel CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/../src/LNLib/*.cpp")
add_library(kernel STATIC ${kernel})
target_compile_definitions(kernel PRIVATE LNLIB_HOME LNLIB_STATIC)
target_include_directories(kernel PUBLIC ../src/LNLib/include PRIVATE ../third_party/eigen)
add_executable(baseline baseline.cpp)
target_link_libraries(baseline PRIVATE kernel)
