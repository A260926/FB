cmake_minimum_required(VERSION 3.16)
project(thb_checks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
file(GLOB_RECURSE CORE CONFIGURE_DEPENDS "${ROOT}/src/LNLib/*.cpp")
add_library(lnlib_offline STATIC ${CORE})
target_include_directories(lnlib_offline PUBLIC "${ROOT}/src/LNLib/include")
target_include_directories(lnlib_offline SYSTEM PRIVATE "${ROOT}/third_party/eigen")
target_compile_definitions(lnlib_offline PUBLIC LNLIB_STATIC PRIVATE LNLIB_HOME)
add_executable(baseline baseline.cpp)
target_link_libraries(baseline PRIVATE lnlib_offline)
if(TASK_TARGET)
 add_executable(target driver.c ../tests/thb_contracts.cpp)
 set_source_files_properties(driver.c PROPERTIES LANGUAGE CXX)
 target_include_directories(target PRIVATE ..)
 target_link_libraries(target PRIVATE lnlib_offline)
 target_compile_options(target PRIVATE -include "${ROOT}/src/LNLib/include/ThbSurface.h")
 add_executable(thb_example ../examples/thb_sculpt.cpp)
 target_link_libraries(thb_example PRIVATE lnlib_offline)
endif()
