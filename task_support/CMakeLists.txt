cmake_minimum_required(VERSION 3.16)
project(opusfile_task LANGUAGES C)
set(CMAKE_C_STANDARD 99)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_DISABLE_INTRINSICS ON CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/vendor/ogg" "${CMAKE_BINARY_DIR}/ogg")
add_subdirectory("${ROOT}/vendor/opus" "${CMAKE_BINARY_DIR}/opus")
file(GLOB PROD CONFIGURE_DEPENDS "${ROOT}/src/*.c")
list(FILTER PROD EXCLUDE REGEX "/(http|wincerts)\\.c$")
add_library(opusfile_task STATIC ${PROD})
target_include_directories(opusfile_task PUBLIC "${ROOT}/include" "${ROOT}/vendor/opus/include")
target_compile_definitions(opusfile_task PRIVATE OP_HAVE_LRINTF=1)
target_link_libraries(opusfile_task PUBLIC opus ogg m)
add_executable(baseline "${CMAKE_CURRENT_SOURCE_DIR}/baseline.c")
target_link_libraries(baseline PRIVATE opusfile_task)
