cmake_minimum_required(VERSION 3.12)
project(task_draco LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_DISABLE_FIND_PACKAGE_Python TRUE CACHE BOOL "No Python" FORCE)
set(DRACO_TESTS OFF CACHE BOOL "Native support only" FORCE)
set(DRACO_TRANSCODER_SUPPORTED OFF CACHE BOOL "No external dependencies" FORCE)
set(DRACO_PMC_SUPPORTED OFF CACHE BOOL "No external dependencies" FORCE)
set(DRACO_BUILD_EXECUTABLES OFF CACHE BOOL "Native support only" FORCE)
set(DRACO_INSTALL OFF CACHE BOOL "No installation" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." "${CMAKE_BINARY_DIR}/draco")
add_executable(baseline baseline.cpp)
target_link_libraries(baseline PRIVATE draco::draco)
target_include_directories(baseline PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../src" "${CMAKE_BINARY_DIR}/draco")
if(TASK_BUILD_TARGET)
  add_executable(task_target driver.c ../task_tests/topology_quantization.cpp)
  set_source_files_properties(driver.c PROPERTIES LANGUAGE CXX)
  set_target_properties(task_target PROPERTIES OUTPUT_NAME target)
  target_include_directories(task_target PRIVATE "${CMAKE_CURRENT_LIST_DIR}/.." "${CMAKE_CURRENT_LIST_DIR}/../src" "${CMAKE_BINARY_DIR}/draco")
  target_link_libraries(task_target PRIVATE draco::draco)
  add_executable(task_example ../examples/topology_quantization.cpp)
  set_target_properties(task_example PROPERTIES OUTPUT_NAME example)
  target_include_directories(task_example PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../src" "${CMAKE_BINARY_DIR}/draco")
  target_link_libraries(task_example PRIVATE draco::draco)
endif()
