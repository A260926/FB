cmake_minimum_required(VERSION 3.16)
project(task_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB_RECURSE CORE_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../source/*.cpp")
list(FILTER CORE_FILES EXCLUDE REGEX "/source/main.cpp$")
add_library(task_core STATIC ${CORE_FILES})
target_include_directories(task_core PUBLIC ../lib/glm ../lib/nlohmann ..)
