cmake_minimum_required(VERSION 3.12)
project(task_draco_session LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_DISABLE_FIND_PACKAGE_Python TRUE CACHE BOOL "Native build" FORCE)
set(DRACO_TESTS OFF CACHE BOOL "Native build" FORCE)
set(DRACO_TRANSCODER_SUPPORTED OFF CACHE BOOL "No optional dependencies" FORCE)
set(DRACO_PMC_SUPPORTED OFF CACHE BOOL "No optional dependencies" FORCE)
set(DRACO_BUILD_EXECUTABLES OFF CACHE BOOL "Native build" FORCE)
set(DRACO_INSTALL OFF CACHE BOOL "Native build" FORCE)
set(SESSION_HEADER "${CMAKE_CURRENT_LIST_DIR}/../src/draco/compression/mesh/mesh_decode_session.h")
set(SESSION_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../src/draco/compression/mesh/mesh_decode_session.cc")
if(TASK_BUILD_TARGET)
  # Force the required public header through actual production compilation.
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -include ${SESSION_HEADER}")
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." "${CMAKE_BINARY_DIR}/draco")
add_executable(baseline baseline.cpp)
target_link_libraries(baseline PRIVATE draco::draco)
target_include_directories(baseline PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../src" "${CMAKE_BINARY_DIR}/draco")
