cmake_minimum_required(VERSION 3.14)
project(surface_anchor_task LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(SUITESPARSE OFF CACHE BOOL "" FORCE)
set(GC_EIGEN_LOCATION "${CMAKE_CURRENT_LIST_DIR}/../.deps/eigen" CACHE PATH "" FORCE)
add_subdirectory(.. gc)
# Preserve the upstream production list and compile any additional surface implementation.
get_target_property(gc_sources geometry-central SOURCES)
file(GLOB surface_sources "${CMAKE_CURRENT_LIST_DIR}/../src/surface/*.cpp")
foreach(source IN LISTS surface_sources)
  file(RELATIVE_PATH relative_source "${CMAKE_CURRENT_LIST_DIR}/../src" "${source}")
  list(FIND gc_sources "${relative_source}" known_index)
  if(known_index EQUAL -1)
    target_sources(geometry-central PRIVATE "${source}")
  endif()
endforeach()
add_executable(task_baseline baseline.cpp)
target_link_libraries(task_baseline PRIVATE geometry-central)
add_executable(task_artifact_check artifact_check.cpp)
