cmake_minimum_required(VERSION 3.12)
project(SheetTaskChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OVM_ENABLE_APPLICATIONS OFF CACHE BOOL "" FORCE)
set(OVM_ENABLE_UNITTESTS OFF CACHE BOOL "" FORCE)
set(OVM_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(OVM_BUILD_DOCUMENTATION OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(.. upstream)
add_executable(baseline baseline.cpp)
target_link_libraries(baseline PRIVATE OpenVolumeMesh)
if(TASK_TARGET)
    set(task_root "${CMAKE_CURRENT_LIST_DIR}/..")
    set(task_header "${task_root}/src/OpenVolumeMesh/Unstable/Topology/SheetCollapse.hh")
    # Force compilation of every required production source, even with old CMake lists.
    target_sources(OpenVolumeMesh PRIVATE "${task_root}/src/OpenVolumeMesh/Unstable/Topology/SheetCollapse.cc")
    add_executable(target "${task_root}/task_tests/sheet_collapse.cpp" driver.c)
    set_source_files_properties(driver.c PROPERTIES LANGUAGE CXX)
    target_include_directories(target PRIVATE "${task_root}")
    target_compile_options(target PRIVATE -include "${task_header}")
    target_link_libraries(target PRIVATE OpenVolumeMesh)
    add_executable(api-consumer api_consumer.cpp)
    target_link_libraries(api-consumer PRIVATE OpenVolumeMesh)
    add_executable(example "${task_root}/examples/sheet_collapse.cc")
    target_include_directories(example PRIVATE "${task_root}")
    target_link_libraries(example PRIVATE OpenVolumeMesh)
endif()
