
cmake_minimum_required(VERSION 3.16)
project(preroll_support C)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
set(SDL_UNIX_CONSOLE_BUILD ON CACHE BOOL "" FORCE)
foreach(opt DIALOG VIDEO RENDER GPU CAMERA JOYSTICK HAPTIC SENSOR ALSA JACK PIPEWIRE PULSEAUDIO SNDIO OSS LIBUDEV DBUS IBUS FCITX COREAUDIO)
 set(SDL_${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(SDL_DUMMYAUDIO ON CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/deps/SDL" sdl)
foreach(opt INSTALL TESTS EXAMPLES VENDORED FLAC FLAC_LIBFLAC FLAC_DRFLAC GME XMP MP3 MIDI OPUS VORBIS_STB VORBIS_VORBISFILE VORBIS_TREMOR WAVPACK)
 set(SDLMIXER_${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(SDLMIXER_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}" mixer)
add_executable(baseline "${ROOT}/task_support/baseline.c")
target_link_libraries(baseline PRIVATE SDL3_mixer::SDL3_mixer-static SDL3::SDL3-static)
if(TASK_TARGET)
 target_sources(SDL3_mixer-static PRIVATE "${ROOT}/src/SDL_mixer_recording.c")
 add_executable(target "${ROOT}/tests/record_preroll.c" "${ROOT}/task_support/driver.c")
 target_include_directories(target PRIVATE "${ROOT}")
 target_link_libraries(target PRIVATE SDL3_mixer::SDL3_mixer-static SDL3::SDL3-static)
 add_executable(record_demo "${ROOT}/examples/record_preroll.c")
 target_link_libraries(record_demo PRIVATE SDL3_mixer::SDL3_mixer-static SDL3::SDL3-static)
endif()
