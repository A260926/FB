cmake_minimum_required(VERSION 3.22)
project(task_cpu LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
foreach(dep fast_float nlohmann_json stb_image tinyexr cgltf)
 add_subdirectory("${ROOT}/exts/${dep}" "deps/${dep}")
endforeach()
file(GLOB CPU_SOURCES CONFIGURE_DEPENDS "${ROOT}/libs/yocto/*.cpp")
list(FILTER CPU_SOURCES EXCLUDE REGEX "yocto_(gui|cutrace)\\.cpp$")
add_library(task_cpu STATIC ${CPU_SOURCES})
target_include_directories(task_cpu PUBLIC "${ROOT}" "${ROOT}/libs/yocto")
target_link_libraries(task_cpu PUBLIC fast_float nlohmann_json::nlohmann_json stb_image::stb_image tinyexr::tinyexr cgltf::cgltf)
find_package(Threads REQUIRED)
target_link_libraries(task_cpu PUBLIC Threads::Threads)
set_source_files_properties(driver.c PROPERTIES LANGUAGE CXX)
add_executable(task_baseline baseline.cpp)
target_link_libraries(task_baseline PRIVATE task_cpu)

if(EXISTS "${ROOT}/checks/compound_lens.cpp")
 add_executable(target "${ROOT}/checks/compound_lens.cpp" driver.c "${ROOT}/libs/yocto/yocto_lens.cpp")
 target_link_libraries(target PRIVATE task_cpu)
 target_compile_options(target PRIVATE -include "${ROOT}/libs/yocto/yocto_lens.h")
endif()
if(EXISTS "${ROOT}/examples/compound_lens.cpp")
 add_executable(compound_lens_example "${ROOT}/examples/compound_lens.cpp" "${ROOT}/libs/yocto/yocto_lens.cpp")
 target_link_libraries(compound_lens_example PRIVATE task_cpu)
 target_compile_options(compound_lens_example PRIVATE -include "${ROOT}/libs/yocto/yocto_lens.h")
endif()
