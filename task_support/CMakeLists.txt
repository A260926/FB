cmake_minimum_required(VERSION 3.14)
project(task_mlt_conform C)
set(CMAKE_C_STANDARD 11)
find_package(Threads REQUIRED)
file(GLOB FRAMEWORK CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../src/framework/*.c")
add_library(task_mlt STATIC ${FRAMEWORK})
target_include_directories(task_mlt PUBLIC ../src "${CMAKE_CURRENT_BINARY_DIR}")
include(GenerateExportHeader)
generate_export_header(task_mlt BASE_NAME MLT EXPORT_FILE_NAME mlt_export.h)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mlt_config.h" "/* POSIX CPU build */\n")
include(CheckFunctionExists)
foreach(fn newlocale uselocale vsnprintf_l)
 string(TOUPPER "${fn}" upper)
 check_function_exists(${fn} HAVE_${upper})
 if(HAVE_${upper})
  target_compile_definitions(task_mlt PRIVATE HAVE_${upper}=1)
 endif()
endforeach()
target_compile_definitions(task_mlt PRIVATE PREFIX_DATA="." PREFIX_LIB=".")
target_link_libraries(task_mlt PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
add_executable(baseline baseline.c)
target_link_libraries(baseline PRIVATE task_mlt)
add_executable(wave_check wave_check.c)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../src/tests/task_conform.c")
 set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/../src/framework/mlt_conform.c" PROPERTIES COMPILE_OPTIONS "-include;${CMAKE_CURRENT_SOURCE_DIR}/../src/framework/mlt_conform.h")
 add_executable(target driver.c ../src/tests/task_conform.c)
 target_include_directories(target PRIVATE ..)
 target_link_libraries(target PRIVATE task_mlt)
 add_executable(binding binding.c)
 target_link_libraries(binding PRIVATE task_mlt)
 add_executable(conform_render ../examples/conform_render.c)
 target_link_libraries(conform_render PRIVATE task_mlt)
endif()
