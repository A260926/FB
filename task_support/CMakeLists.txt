cmake_minimum_required(VERSION 3.16)
project(reference_flow_support LANGUAGES C CXX)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_LIBWASM OFF CACHE BOOL "" FORCE)
set(WABT_INSTALL_RULES OFF CACHE BOOL "" FORCE)
set(USE_INTERNAL_SHA256 ON CACHE BOOL "" FORCE)
set(WITH_WASI OFF CACHE BOOL "" FORCE)
add_subdirectory(.. wabt)
get_target_property(known wabt SOURCES)
set(absolute_known)
foreach(path IN LISTS known)
  get_filename_component(path "${path}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/..")
  list(APPEND absolute_known "${path}")
endforeach()
file(GLOB candidates CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../src/*.cc")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/upstream-root-sources.txt" upstream_names)
foreach(path IN LISTS candidates)
  get_filename_component(name "${path}" NAME)
  if(NOT path IN_LIST absolute_known AND NOT name IN_LIST upstream_names)
    target_sources(wabt PRIVATE "${path}")
  endif()
endforeach()
add_executable(task_baseline baseline.cc)
target_link_libraries(task_baseline PRIVATE wabt)
target_include_directories(task_baseline PRIVATE ..)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../test/task_reference_flow.cc")
  add_executable(task_target driver.c ../test/task_reference_flow.cc)
  set_source_files_properties(driver.c PROPERTIES LANGUAGE CXX)
  target_include_directories(task_target PRIVATE ..)
  target_link_libraries(task_target PRIVATE wabt)
  set_target_properties(task_target PROPERTIES OUTPUT_NAME target RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.task-build")
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../src/tools/wasm-ref-flow.cc")
  add_executable(wasm-ref-flow ../src/tools/wasm-ref-flow.cc)
  target_link_libraries(wasm-ref-flow PRIVATE wabt)
endif()
