cmake_minimum_required(VERSION 3.30)
project(ozz LANGUAGES C CXX)
set(ozz_build_tools OFF CACHE BOOL "" FORCE)
set(ozz_build_fbx OFF CACHE BOOL "" FORCE)
set(ozz_build_gltf OFF CACHE BOOL "" FORCE)
set(ozz_build_samples OFF CACHE BOOL "" FORCE)
set(ozz_build_howtos OFF CACHE BOOL "" FORCE)
set(ozz_build_tests OFF CACHE BOOL "" FORCE)
set(ozz_build_data OFF CACHE BOOL "" FORCE)
add_subdirectory(.. ozz)
get_filename_component(SEED_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
if(TASK_MODE STREQUAL "target")
  target_sources(ozz_animation_offline PRIVATE "${SEED_ROOT}/src/animation/offline/skeleton_reducer.cc")
  add_executable(task_runner driver.cpp ../test/animation/offline/skeleton_reducer_target.cpp)
else()
  add_executable(task_runner baseline.cpp)
endif()
target_compile_features(task_runner PRIVATE cxx_std_17)
target_include_directories(task_runner PRIVATE "${SEED_ROOT}" "${SEED_ROOT}/include")
target_link_libraries(task_runner PRIVATE ozz_animation_offline ozz_geometry)
