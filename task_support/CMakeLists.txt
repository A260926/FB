cmake_minimum_required(VERSION 3.16)
project(sisl_imprint_cpu LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
include_directories(${ROOT}/include ${ROOT})
file(GLOB C_CORE CONFIGURE_DEPENDS ${ROOT}/src/*.c)
file(GLOB CXX_CORE CONFIGURE_DEPENDS ${ROOT}/src/*.cpp ${ROOT}/src/*.cxx ${ROOT}/src/*.C)
add_library(sisl STATIC ${C_CORE} ${CXX_CORE})
add_executable(sisl_baseline ${ROOT}/task_support/baseline.cpp)
target_link_libraries(sisl_baseline PRIVATE sisl)
if(TASK_TARGET)
 set_source_files_properties(${ROOT}/task_support/driver.c PROPERTIES LANGUAGE CXX)
 add_executable(target ${ROOT}/tests/imprint.cpp ${ROOT}/task_support/driver.c ${ROOT}/src/sisl_imprint.cpp ${ROOT}/src/s1310_events.c)
 target_link_libraries(target PRIVATE sisl)
 add_executable(imprint_obj ${ROOT}/examples/imprint_obj.cpp)
 target_link_libraries(imprint_obj PRIVATE sisl)
endif()
