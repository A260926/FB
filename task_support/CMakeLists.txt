cmake_minimum_required(VERSION 3.16)
project(arib_macro_task LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(ARIBCC_NO_RENDERER ON CACHE BOOL "decoder only" FORCE)
set(ARIBCC_BUILD_TESTS OFF CACHE BOOL "custom tests" FORCE)
add_subdirectory(.. arib)
add_executable(task_baseline baseline.cpp)
target_include_directories(task_baseline PRIVATE ..)
target_link_libraries(task_baseline PRIVATE aribcaption)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../test/decode/user_macro.cpp")
 add_executable(task_target driver.cpp ../test/decode/user_macro.cpp)
 target_include_directories(task_target PRIVATE ..)
 target_link_libraries(task_target PRIVATE aribcaption)
endif()
