cmake_minimum_required(VERSION 3.16)
project(loop_check C)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
foreach(feature SHARED TEST_LIBRARY TESTS EXAMPLES INSTALL RENDER GPU CAMERA JOYSTICK HAPTIC HIDAPI SENSOR POWER DIALOG COREAUDIO ALSA JACK PULSEAUDIO SNDIO OSS LIBUDEV X11 WAYLAND OPENGL OPENGLES VULKAN KMSDRM)
  set(SDL_${feature} OFF CACHE BOOL "fixed headless source build" FORCE)
endforeach()
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_VIDEO ON CACHE BOOL "Mac SDL3 tray implementation needs video compilation; no window/device opened" FORCE)
set(SDL_DUMMYVIDEO ON CACHE BOOL "" FORCE)
set(SDL_AUDIO ON CACHE BOOL "" FORCE)
set(SDL_DUMMYAUDIO ON CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/vendor/SDL" sdl EXCLUDE_FROM_ALL)
add_library(task_sound STATIC "${ROOT}/src/SDL_sound.c" "${ROOT}/src/SDL_sound_wav.c" "${ROOT}/src/SDL_sound_loop.c")
target_include_directories(task_sound PUBLIC "${ROOT}/include")
target_compile_definitions(task_sound PRIVATE SOUND_SUPPORTS_WAV=1 SOUND_SUPPORTS_MP3=0 SOUND_SUPPORTS_MIDI=0 SOUND_SUPPORTS_MODPLUG=0 SOUND_SUPPORTS_AIFF=0 SOUND_SUPPORTS_AU=0 SOUND_SUPPORTS_VORBIS=0 SOUND_SUPPORTS_VOC=0 SOUND_SUPPORTS_RAW=0 SOUND_SUPPORTS_SHN=0 SOUND_SUPPORTS_FLAC=0 SOUND_SUPPORTS_COREAUDIO=0)
target_link_libraries(task_sound PUBLIC SDL3::SDL3-static)
add_executable(native-baseline baseline.c)
target_include_directories(native-baseline PRIVATE "${ROOT}")
target_link_libraries(native-baseline PRIVATE task_sound)
if(TASK_TARGET)
  target_compile_options(task_sound PRIVATE -include "${ROOT}/src/SDL_sound_loop.h")
  add_executable(api-probe api_probe.c)
  add_executable(loop-example "${ROOT}/examples/loop_region.c")
  add_executable(task-target "${ROOT}/tests/loop_region.c" driver.c)
  add_executable(output-probe output_probe.c)
  foreach(prog api-probe loop-example task-target output-probe)
    target_include_directories(${prog} PRIVATE "${ROOT}")
    target_link_libraries(${prog} PRIVATE task_sound)
  endforeach()
endif()
