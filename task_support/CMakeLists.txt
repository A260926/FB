cmake_minimum_required(VERSION 3.22)
project(task_cpu LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
file(GLOB CPU_SOURCES CONFIGURE_DEPENDS "${ROOT}/src/*.c")
add_library(task_cpu STATIC ${CPU_SOURCES})
target_include_directories(task_cpu PUBLIC "${ROOT}/include" "${ROOT}")
target_link_libraries(task_cpu PUBLIC m)
target_compile_definitions(task_cpu PRIVATE __STDC_WANT_LIB_EXT1__=0)
add_executable(task_baseline baseline.c)
target_link_libraries(task_baseline PRIVATE task_cpu)
if(EXISTS "${ROOT}/checks/shader_vm.c")
 add_executable(target "${ROOT}/checks/shader_vm.c" driver.c "${ROOT}/src/shader.c" "${ROOT}/src/shader_vm.c")
 target_link_libraries(target PRIVATE task_cpu)
 target_compile_options(target PRIVATE -include "${ROOT}/include/TGL/glshader.h")
endif()
if(EXISTS "${ROOT}/examples/shader_vm.c")
 add_executable(shader_vm_example "${ROOT}/examples/shader_vm.c" "${ROOT}/src/shader.c" "${ROOT}/src/shader_vm.c")
 target_link_libraries(shader_vm_example PRIVATE task_cpu)
 target_compile_options(shader_vm_example PRIVATE -include "${ROOT}/include/TGL/glshader.h")
endif()
