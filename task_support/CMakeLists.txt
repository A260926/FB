cmake_minimum_required(VERSION 3.16)
project(audiofile_task LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
include(TestBigEndian)
test_big_endian(IS_BIG_ENDIAN)
if(IS_BIG_ENDIAN)
  set(ENDIAN_VALUE 1)
else()
  set(ENDIAN_VALUE 0)
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.h" "#define WORDS_BIGENDIAN ${ENDIAN_VALUE}\n#define HAVE_UNISTD_H 1\n#define ENABLE_FLAC 0\n")
file(GLOB_RECURSE PROD CONFIGURE_DEPENDS "${ROOT}/libaudiofile/*.cpp" "${ROOT}/libaudiofile/*.c")
list(FILTER PROD EXCLUDE REGEX "/UT_[^/]+\\.cpp$")
add_library(audiofile_task STATIC ${PROD})
target_include_directories(audiofile_task PUBLIC "${ROOT}/libaudiofile" "${ROOT}" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${ROOT}/libaudiofile/alac")
target_compile_options(audiofile_task PRIVATE -Wno-deprecated-declarations -Wno-expansion-to-defined)
add_executable(baseline "${CMAKE_CURRENT_SOURCE_DIR}/baseline.cpp")
target_link_libraries(baseline PRIVATE audiofile_task m)
