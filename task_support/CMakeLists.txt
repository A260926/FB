cmake_minimum_required(VERSION 3.16)
project(IsoSplitChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SOURCE "${CMAKE_CURRENT_LIST_DIR}/..")
add_subdirectory("${SOURCE}" upstream)
add_executable(task_baseline baseline.cpp)
target_include_directories(task_baseline PRIVATE "${SOURCE}")
target_link_libraries(task_baseline PRIVATE opennurbsStatic)
if(EXISTS "${SOURCE}/tests/iso_split_contracts.cpp")
  if(NOT EXISTS "${SOURCE}/opennurbs_brep_isosplit.cpp" OR NOT EXISTS "${SOURCE}/examples/iso_split.cpp")
    message(FATAL_ERROR "Production implementation and example are required")
  endif()
  target_sources(opennurbsStatic PRIVATE "${SOURCE}/opennurbs_brep_isosplit.cpp")
  set_source_files_properties(driver.c PROPERTIES LANGUAGE CXX)
  add_executable(task_target driver.c "${SOURCE}/tests/iso_split_contracts.cpp")
  set_target_properties(task_target PROPERTIES OUTPUT_NAME target)
  target_include_directories(task_target PRIVATE "${SOURCE}")
  target_link_libraries(task_target PRIVATE opennurbsStatic)
  add_executable(task_example "${SOURCE}/examples/iso_split.cpp")
  target_include_directories(task_example PRIVATE "${SOURCE}")
  target_link_libraries(task_example PRIVATE opennurbsStatic)
endif()
