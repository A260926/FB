cmake_minimum_required(VERSION 3.5)
project(ordered_support LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
get_filename_component(MATROSKA_TASK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${MATROSKA_TASK_ROOT}/third_party/libebml" ebml)
target_include_directories(ebml INTERFACE "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/ebml>")
add_library(EBML::ebml ALIAS ebml)
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}")
add_subdirectory("${MATROSKA_TASK_ROOT}" matroska)
target_include_directories(matroska INTERFACE "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/matroska>")
add_executable(ordered_baseline ../baseline.cpp)
target_link_libraries(ordered_baseline PRIVATE matroska)
