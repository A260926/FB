cmake_minimum_required(VERSION 3.12)
project(ovm_baseline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(OVM_ENABLE_APPLICATIONS OFF CACHE BOOL "" FORCE)
set(OVM_ENABLE_UNITTESTS OFF CACHE BOOL "" FORCE)
set(OVM_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(OVM_BUILD_DOCUMENTATION OFF CACHE BOOL "" FORCE)
add_subdirectory(.. ovm)
add_executable(boundary_baseline boundary_baseline.cc)
target_link_libraries(boundary_baseline PRIVATE OpenVolumeMesh::OpenVolumeMesh)
add_executable(check_selection_boundary check_selection_boundary.cc)
target_link_libraries(check_selection_boundary PRIVATE OpenVolumeMesh::OpenVolumeMesh)
