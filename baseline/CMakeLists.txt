cmake_minimum_required(VERSION 3.13)
project(UserFBStartingPoint LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(LPM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(LIB_PROTO_MUTATOR_TESTING OFF CACHE BOOL "" FORCE)
set(LIB_PROTO_MUTATOR_EXAMPLES OFF CACHE BOOL "" FORCE)
set(protobuf_MODULE_COMPATIBLE ON)
find_package(Protobuf 3.21.12 EXACT REQUIRED)
set(CMAKE_DISABLE_FIND_PACKAGE_LibLZMA ON)
set(CMAKE_DISABLE_FIND_PACKAGE_ZLIB ON)
add_subdirectory("${LPM_ROOT}" lpm)
# LZMA may add a system prefix containing a different protobuf; pinned headers win.
target_include_directories(protobuf-mutator BEFORE PRIVATE ${Protobuf_INCLUDE_DIRS})
target_include_directories(protobuf-mutator-libfuzzer BEFORE PRIVATE ${Protobuf_INCLUDE_DIRS})
add_subdirectory("${LPM_ROOT}/.deps/src/googletest" googletest EXCLUDE_FROM_ALL)
set(PROTO_FILES src/mutator_test_proto2.proto src/mutator_test_proto3.proto)
set(PROTO_SRCS)
foreach(file IN LISTS PROTO_FILES)
  string(REPLACE ".proto" ".pb.cc" generated "${file}")
  string(REPLACE ".proto" ".pb.h" header "${file}")
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${generated}" "${CMAKE_CURRENT_BINARY_DIR}/${header}"
    COMMAND "${Protobuf_PROTOC_EXECUTABLE}" --cpp_out=${CMAKE_CURRENT_BINARY_DIR} -I${LPM_ROOT} -I${Protobuf_INCLUDE_DIR} "${LPM_ROOT}/${file}"
    DEPENDS "${LPM_ROOT}/${file}")
  list(APPEND PROTO_SRCS "${CMAKE_CURRENT_BINARY_DIR}/${generated}")
endforeach()
add_library(baseline_proto STATIC ${PROTO_SRCS})
target_include_directories(baseline_proto PUBLIC "${CMAKE_CURRENT_BINARY_DIR}" ${Protobuf_INCLUDE_DIRS})
target_link_libraries(baseline_proto PUBLIC ${Protobuf_LIBRARIES})
add_executable(baseline baseline.cc)
target_include_directories(baseline PRIVATE "${LPM_ROOT}")
target_link_libraries(baseline PRIVATE protobuf-mutator baseline_proto)
add_executable(old_mutator_tests "${LPM_ROOT}/src/mutator_test.cc" "${LPM_ROOT}/src/utf8_fix_test.cc" "${LPM_ROOT}/src/weighted_reservoir_sampler_test.cc")
target_include_directories(old_mutator_tests PRIVATE "${LPM_ROOT}")
target_link_libraries(old_mutator_tests PRIVATE protobuf-mutator baseline_proto gtest_main gmock)
# Build the optional dynamic Any regression source when available.
if(EXISTS "${LPM_ROOT}/src/dynamic_any_test.cc")
  add_executable(dynamic_any_tests "${LPM_ROOT}/src/dynamic_any_test.cc")
  target_include_directories(dynamic_any_tests PRIVATE "${LPM_ROOT}")
  target_link_libraries(dynamic_any_tests PRIVATE protobuf-mutator baseline_proto gtest_main gmock)
endif()
