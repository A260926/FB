cmake_minimum_required(VERSION 3.13)
project(leveldb_checks CXX)
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
file(GLOB candidates "${root}/db/*.cc" "${root}/table/*.cc" "${root}/util/*.cc")
foreach(f IN LISTS candidates)
 get_filename_component(name "${f}" NAME)
 if(NOT name MATCHES "test|bench|leveldbutil")
  list(APPEND core "${f}")
 endif()
endforeach()
add_library(leveldb_checks STATIC ${core} "${root}/port/port_posix.cc" "${root}/port/port_posix_sse.cc")
set(CMAKE_CXX_STANDARD 11)
set_property(TARGET leveldb_checks PROPERTY CXX_STANDARD 11)
target_include_directories(leveldb_checks PUBLIC "${root}" "${root}/include")
target_compile_definitions(leveldb_checks PUBLIC LEVELDB_PLATFORM_POSIX LEVELDB_ATOMIC_PRESENT)
if(APPLE)
 target_compile_definitions(leveldb_checks PUBLIC OS_MACOSX)
else()
 target_compile_definitions(leveldb_checks PUBLIC OS_LINUX)
endif()
find_package(Threads REQUIRED)
target_link_libraries(leveldb_checks PUBLIC Threads::Threads)
add_library(routing_fixture STATIC routing_fixture.cpp)
target_link_libraries(routing_fixture PUBLIC leveldb_checks)
add_executable(routing_baseline routing_baseline.cpp)
target_link_libraries(routing_baseline PRIVATE routing_fixture)
add_executable(baseline baseline.cpp)
target_link_libraries(baseline PRIVATE leveldb_checks)
add_executable(legacy_filter "${root}/table/filter_block_test.cc" "${root}/util/testharness.cc" "${root}/util/testutil.cc")
target_link_libraries(legacy_filter PRIVATE leveldb_checks)
if(EXISTS "${root}/tests/multiget.cpp")
 add_executable(target target_main.cpp "${root}/tests/multiget.cpp")
 target_include_directories(target PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
 target_link_libraries(target PRIVATE routing_fixture)
endif()
