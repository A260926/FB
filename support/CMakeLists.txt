cmake_minimum_required(VERSION 3.30)
project(TaskSupport LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../src/smooth/ftgrays.c" gray_source)
if(TASK_MUTATION STREQUAL "drop_band")
 string(REPLACE "ras.min_ey    = band[1];" "ras.min_ey    = band[1] + 1;" gray_source "${gray_source}")
elseif(TASK_MUTATION STREQUAL "lose_parity")
 string(REPLACE "if ( ras.outline.flags & FT_OUTLINE_EVEN_ODD_FILL )" "if ( 0 && (ras.outline.flags & FT_OUTLINE_EVEN_ODD_FILL) )" gray_source "${gray_source}")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gray-default.c" "${gray_source}")
string(REPLACE "#define FT_RENDER_POOL_SIZE  16384L" "#define FT_RENDER_POOL_SIZE  2048L" gray_source "${gray_source}")
string(REPLACE "ft_grays_raster" "ft_grays_raster_small" gray_source "${gray_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gray-small.c" "${gray_source}")
add_library(gray_default STATIC "${CMAKE_CURRENT_BINARY_DIR}/gray-default.c")
add_library(gray_small STATIC "${CMAKE_CURRENT_BINARY_DIR}/gray-small.c")
foreach(t gray_default gray_small)
 target_compile_definitions(${t} PUBLIC STANDALONE_)
 target_compile_options(${t} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/trace_compat.h")
 target_include_directories(${t} PUBLIC ../include/freetype ../src/smooth)
endforeach()
add_executable(task_baseline baseline.cpp)
target_include_directories(task_baseline PRIVATE ../include ..)
target_link_libraries(task_baseline PRIVATE gray_default gray_small)
if(TASK_TARGET)
 add_executable(task_target runner.cpp target.cpp)
 target_include_directories(task_target PRIVATE . ../include ..)
 target_link_libraries(task_target PRIVATE gray_default gray_small)
endif()
