cmake_minimum_required(VERSION 3.16)
project(SurfaceJoinOffline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
file(GLOB_RECURSE CORE CONFIGURE_DEPENDS "${ROOT}/src/LNLib/*.cpp")
add_library(lnlib_offline STATIC ${CORE})
target_include_directories(lnlib_offline PUBLIC "${ROOT}/src/LNLib/include")
target_include_directories(lnlib_offline SYSTEM PRIVATE "${ROOT}/third_party/eigen")
target_compile_definitions(lnlib_offline PUBLIC LNLIB_STATIC PRIVATE LNLIB_HOME)
add_executable(baseline baseline.cpp)
target_link_libraries(baseline PRIVATE lnlib_offline)
option(JOIN_TARGET "Compile new surface join tests and example" OFF)
if(JOIN_TARGET)
  add_executable(join_contracts driver.cpp "${ROOT}/tests/T_SurfaceJoin.cpp")
  target_include_directories(join_contracts PRIVATE "${ROOT}")
  target_link_libraries(join_contracts PRIVATE lnlib_offline)
  add_executable(join_example "${ROOT}/examples/join_surface.cpp")
  target_link_libraries(join_example PRIVATE lnlib_offline)
endif()
